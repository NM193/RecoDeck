**Rust: get_track_artwork command in library.rs:**

Add a new Tauri command that extracts embedded artwork from audio files using lofty (already installed at v0.22):

```rust
use tauri::ipc::Response;

#[tauri::command]
pub async fn get_track_artwork(
    track_id: i64,
    state: State<'_, AppState>,
) -> Result<Response, String> {
    let db = state.db.lock().unwrap();
    let db = db.as_ref().ok_or("Database not initialized")?;
    let track = db.get_track(track_id).map_err(|e| e.to_string())?;

    let file_path = &track.file_path;

    // Try embedded art via lofty
    if let Ok(tagged_file) = lofty::read_from_path(file_path) {
        // Use primary_tag() for format-appropriate tag, then fall back to all tags
        for tag in tagged_file.tags() {
            if let Some(picture) = tag.pictures().first() {
                return Ok(Response::new(picture.data().to_vec()));
            }
        }
    }

    // Try folder cover images
    if let Some(folder) = std::path::Path::new(file_path).parent() {
        for name in &["cover.jpg", "cover.png", "folder.jpg", "folder.png", "Cover.jpg", "Cover.png", "Folder.jpg", "Folder.png"] {
            let cover_path = folder.join(name);
            if cover_path.exists() {
                if let Ok(bytes) = std::fs::read(&cover_path) {
                    return Ok(Response::new(bytes));
                }
            }
        }
    }

    Err("no_artwork".to_string())
}
```

```typescript
// Module-level cache — NOT React state, NOT Zustand store
const artworkCache = new Map<number, string | null>()

export async function getTrackArtworkUrl(trackId: number): Promise<string | null> {
  if (artworkCache.has(trackId)) return artworkCache.get(trackId)!
  try {
    const buffer = await tauriApi.getTrackArtwork(trackId)
    const bytes = new Uint8Array(buffer)
    const blob = new Blob([bytes])
    const url = URL.createObjectURL(blob)
    artworkCache.set(trackId, url)
    return url
  } catch {
    artworkCache.set(trackId, null)
    return null
  }
}
```

If `invoke<ArrayBuffer>` doesn't return an ArrayBuffer (Tauri v2 binary response issue from research open question), adapt to receive a base64 string and convert:
```typescript
const base64 = await invoke<string>('get_track_artwork', { trackId })
const url = `data:image/jpeg;base64,${base64}`
```
