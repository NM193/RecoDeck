**Step 1: Check if the DB has a search function that includes analysis data.**

The current `search_tracks` in `db/mod.rs` returns `Vec<Track>` without analysis. Two approaches:

**Option A (preferred — simpler):** After getting search results, look up analysis data for each result. This avoids adding a new DB method and keeps the search function simple.

In `src-tauri/src/server/routes.rs`, update the `search_tracks` handler:

```rust
async fn search_tracks(
    State(state): State<Arc<CompanionServerState>>,
    Query(params): Query<SearchParams>,
) -> Result<Json<Vec<MobileTrackDTO>>, StatusCode> {
    let query = params.q.unwrap_or_default();
    if query.is_empty() {
        return Ok(Json(Vec::new()));
    }

    let db_lock = state.db.lock().map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;
    let db = db_lock.as_ref().ok_or(StatusCode::SERVICE_UNAVAILABLE)?;

    let tracks = db
        .search_tracks(&query)
        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

    let mobile_tracks: Vec<MobileTrackDTO> = tracks
        .into_iter()
        .map(|track| {
            let track_id = track.id.unwrap_or(0);
            // Look up analysis data for each search result
            let (bpm, key) = db
                .get_track_analysis(track_id)
                .ok()
                .map(|a| (a.0, a.2))
                .unwrap_or((None, None));
            MobileTrackDTO::from_track_with_analysis(track, bpm, key)
        })
        .collect();

    Ok(Json(mobile_tracks))
}
```
