**Step 2: Create a retry helper for rate-limited errors.**

Add this helper function INSIDE the `create<AIState>((set, get) => ({` block, as a private utility used by `sendMessage` and `generatePlaylist`. Since Zustand stores are plain objects, define it as a standalone function ABOVE the store definition:

```typescript
/** Rate-limit retry helper. Retries the operation up to 2 times with 30s backoff. */
async function withRateLimitRetry(
  operation: () => Promise<void>,
  setError: (error: string | null) => void,
  setGenerating: (isGenerating: boolean) => void,
): Promise<void> {
  try {
    await operation();
  } catch (e) {
    if (isAppError(e) && e.kind === 'AiNetwork' && e.message?.includes('Rate limited')) {
      for (let attempt = 1; attempt <= 2; attempt++) {
        setError(`Rate limited \u2014 retrying in 30 seconds... (attempt ${attempt}/2)`);
        await new Promise((resolve) => setTimeout(resolve, 30_000));
        try {
          await operation();
          setError(null);
          return;
        } catch (retryErr) {
          if (attempt === 2) {
            setError(getErrorMessage(retryErr));
            setGenerating(false);
          }
        }
      }
    } else {
      throw e;
    }
  }
}
```

```rust
/// Get MIME type for an audio file based on its extension.
/// Shared across stream:// protocol handler (lib.rs) and companion server streaming (streaming.rs).
pub fn audio_mime_type(path: &str) -> &'static str {
    match std::path::Path::new(path)
        .extension()
        .and_then(|e| e.to_str())
        .map(|s| s.to_lowercase())
        .as_deref()
    {
        Some("mp3") => "audio/mpeg",
        Some("flac") => "audio/flac",
        Some("wav") => "audio/wav",
        Some("ogg") => "audio/ogg",
        Some("m4a") => "audio/mp4",
        Some("aac") => "audio/aac",
        Some("aiff") | Some("aif") => "audio/aiff",
        _ => "application/octet-stream",
    }
}
```
