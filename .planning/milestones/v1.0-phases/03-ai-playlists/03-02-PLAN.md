**Camelot compatibility function (inline in the component):**
```typescript
function getKeyCompatibility(keyA: string | undefined, keyB: string | undefined): 'perfect' | 'compatible' | 'clash' {
  if (!keyA || !keyB) return 'clash';
  if (keyA === keyB) return 'perfect';
  const parse = (k: string) => {
    const m = k.match(/^(\d{1,2})([AB])$/i);
    if (!m) return null;
    return { num: parseInt(m[1]), letter: m[2].toUpperCase() };
  };
  const a = parse(keyA);
  const b = parse(keyB);
  if (!a || !b) return 'clash';
  // Same number, different letter (inner/outer)
  if (a.num === b.num && a.letter !== b.letter) return 'compatible';
  // Same letter, ±1 (circular 1-12)
  if (a.letter === b.letter) {
    const diff = Math.abs(a.num - b.num);
    if (diff === 1 || diff === 11) return 'compatible';
  }
  return 'clash';
}
```
