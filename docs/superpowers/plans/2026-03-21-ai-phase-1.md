pub fn search_conversations(&self, query: &str, limit: usize) -> Result<Vec<(String, String, Vec<ConversationMessage>)>> {
    // Find conversations containing the query in message content
    let search_pattern = format!("%{}%", query);
    let mut stmt = self.conn.prepare(
        "SELECT DISTINCT m.conversation_id, c.title
         FROM ai_messages m
         JOIN ai_conversations c ON c.id = m.conversation_id
         WHERE m.content LIKE ?1
         ORDER BY c.created_at DESC
         LIMIT ?2"
    )?;
    let convos: Vec<(String, String)> = stmt.query_map(params![search_pattern, limit], |row| {
        Ok((row.get::<_, String>(0)?, row.get::<_, String>(1)?))
    })?.collect::<std::result::Result<Vec<_>, _>>()?;

    let mut results = Vec::new();
    for (conv_id, title) in convos {
        // Get relevant messages from this conversation (those matching + surrounding context)
        let mut msg_stmt = self.conn.prepare(
            "SELECT id, conversation_id, role, content, metadata_json, created_at
             FROM ai_messages
             WHERE conversation_id = ?1 AND content LIKE ?2
             ORDER BY created_at ASC
             LIMIT 6"
        )?;
        let messages: Vec<ConversationMessage> = msg_stmt.query_map(params![conv_id, search_pattern], |row| {
            Ok(ConversationMessage {
                id: row.get(0)?,
                conversation_id: row.get(1)?,
                role: row.get(2)?,
                content: row.get(3)?,
                metadata_json: row.get(4)?,
                created_at: row.get(5)?,
            })
        })?.collect::<std::result::Result<Vec<_>, _>>()?;
        results.push((conv_id, title, messages));
    }
    Ok(results)
}

pub fn search_tracks_filtered(
    &self,
    query: Option<&str>,
    artist: Option<&str>,
    genre: Option<&str>,
    bpm_min: Option<f64>,
    bpm_max: Option<f64>,
    key: Option<&str>,
    tags: &[String],
    limit: usize,
) -> Result<Vec<(Track, Option<f64>, Option<String>)>> {
    let mut sql = String::from(
        "SELECT t.*, a.bpm, a.musical_key FROM tracks t
         LEFT JOIN track_analysis a ON a.track_id = t.id"
    );
    let mut conditions = Vec::new();
    let mut param_values: Vec<Box<dyn rusqlite::types::ToSql>> = Vec::new();

    if let Some(q) = query {
        let idx = param_values.len() + 1;
        conditions.push(format!("(t.title LIKE ?{} OR t.artist LIKE ?{} OR t.album LIKE ?{})", idx, idx, idx));
        let pattern = format!("%{}%", q);
        param_values.push(Box::new(pattern));
    }
    if let Some(a) = artist {
        conditions.push(format!("t.artist LIKE ?{}", param_values.len() + 1));
        param_values.push(Box::new(format!("%{}%", a)));
    }
    if let Some(g) = genre {
        conditions.push(format!(
            "t.id IN (SELECT track_id FROM track_genres WHERE genre LIKE ?{})",
            param_values.len() + 1
        ));
        param_values.push(Box::new(format!("%{}%", g)));
    }
    if let Some(min) = bpm_min {
        conditions.push(format!("a.bpm >= ?{}", param_values.len() + 1));
        param_values.push(Box::new(min));
    }
    if let Some(max) = bpm_max {
        conditions.push(format!("a.bpm <= ?{}", param_values.len() + 1));
        param_values.push(Box::new(max));
    }
    if let Some(k) = key {
        conditions.push(format!("a.musical_key = ?{}", param_values.len() + 1));
        param_values.push(Box::new(k.to_string()));
    }
    if !tags.is_empty() {
        for tag in tags {
            conditions.push(format!(
                "t.id IN (SELECT tt.track_id FROM track_tags tt JOIN tags tg ON tg.id = tt.tag_id WHERE tg.name = ?{})",
                param_values.len() + 1
            ));
            param_values.push(Box::new(tag.clone()));
        }
    }

    if !conditions.is_empty() {
        sql.push_str(" WHERE ");
        sql.push_str(&conditions.join(" AND "));
    }
    sql.push_str(&format!(" LIMIT {}", limit));

    let mut stmt = self.conn.prepare(&sql)?;
    let params_refs: Vec<&dyn rusqlite::types::ToSql> = param_values.iter().map(|p| p.as_ref()).collect();
    // NOTE: The Track struct construction below must include ALL fields from the actual Track struct
    // in db/mod.rs (lines 39-63). Check the current struct definition and include any fields
    // like genre, genre_source, track_number that may have been added. Use row.get("column_name")?
    // for required fields and row.get("column_name").ok().flatten() for optional ones.

            id: row.get("id")?,
            file_path: row.get("file_path")?,
            file_hash: row.get("file_hash")?,
            title: row.get("title")?,
            artist: row.get("artist")?,
            album: row.get("album")?,

            year: row.get("year")?,
            label: row.get("label")?,
            duration_ms: row.get("duration_ms")?,
            file_format: row.get("file_format")?,
            bitrate: row.get("bitrate")?,
            sample_rate: row.get("sample_rate")?,
            file_size: row.get("file_size")?,
            date_added: row.get("date_added")?,
            date_modified: row.get("date_modified")?,
            play_count: row.get("play_count")?,
            rating: row.get("rating")?,
            comment: row.get("comment")?,
            artwork_path: row.get("artwork_path")?,
            // Include ALL remaining Track fields from the struct definition

use serde::Serialize;
use crate::db::Database;

#[derive(Serialize)]
pub struct TasteProfile {
    pub total_tracks: i64,
    pub top_genres: Vec<GenreEntry>,
    pub top_artists: Vec<ArtistEntry>,
    pub bpm_range: BpmRange,
    pub most_played: Vec<TrackEntry>,
    pub top_rated: Vec<TrackEntry>,
    pub top_tags: Vec<TagEntry>,
}

#[derive(Serialize)]
pub struct GenreEntry {
    pub genre: String,
    pub count: i64,
}

#[derive(Serialize)]
pub struct ArtistEntry {
    pub artist: String,
    pub count: i64,
}

#[derive(Serialize)]
pub struct BpmRange {
    pub min: f64,
    pub max: f64,
    pub avg: f64,
}

#[derive(Serialize)]
pub struct TrackEntry {
    pub title: String,
    pub artist: String,
}

#[derive(Serialize)]
pub struct TagEntry {
    pub name: String,
    pub category: String,
    pub count: i64,
}

pub fn build_taste_profile(db: &Database) -> Result<String, String> {
    let total_tracks = db.count_tracks().map_err(|e| e.to_string())?;

    let top_genres: Vec<GenreEntry> = db.get_top_genres(10)
        .map_err(|e| e.to_string())?
        .into_iter()
        .map(|(genre, count)| GenreEntry { genre, count })
        .collect();

    let top_artists: Vec<ArtistEntry> = db.get_top_artists(10)
        .map_err(|e| e.to_string())?
        .into_iter()
        .map(|(artist, count)| ArtistEntry { artist, count })
        .collect();

    let (bpm_min, bpm_max, bpm_avg) = db.get_bpm_distribution()
        .map_err(|e| e.to_string())?;
    let bpm_range = BpmRange {
        min: bpm_min.unwrap_or(0.0),
        max: bpm_max.unwrap_or(0.0),
        avg: bpm_avg.unwrap_or(0.0),
    };

    let most_played: Vec<TrackEntry> = db.get_most_played_tracks(10)
        .map_err(|e| e.to_string())?
        .into_iter()
        .map(|(_, title, artist, _)| TrackEntry { title, artist })
        .collect();

    let top_rated: Vec<TrackEntry> = db.get_top_rated_tracks(10)
        .map_err(|e| e.to_string())?
        .into_iter()
        .map(|(_, title, artist, _)| TrackEntry { title, artist })
        .collect();

    let top_tags: Vec<TagEntry> = db.get_tag_usage(10)
        .map_err(|e| e.to_string())?
        .into_iter()
        .map(|(name, category, count)| TagEntry { name, category, count })
        .collect();

    let profile = TasteProfile {
        total_tracks,
        top_genres,
        top_artists,
        bpm_range,
        most_played,
        top_rated,
        top_tags,
    };

    serde_json::to_string(&profile).map_err(|e| e.to_string())
}

fn execute_search_library(db: &Database, input: &Value) -> (String, ActionResult) {
    let query = input.get("query").and_then(|v| v.as_str());
    let artist = input.get("artist").and_then(|v| v.as_str());
    let genre = input.get("genre").and_then(|v| v.as_str());
    let bpm_range = input.get("bpm_range").and_then(|v| v.as_array());
    let key = input.get("key").and_then(|v| v.as_str());
    let tags: Vec<String> = input.get("tags")
        .and_then(|v| v.as_array())
        .map(|arr| arr.iter().filter_map(|v| v.as_str().map(String::from)).collect())
        .unwrap_or_default();
    let limit = input.get("limit").and_then(|v| v.as_u64()).unwrap_or(20) as usize;

    let bpm_min = bpm_range.and_then(|r| r.first()).and_then(|v| v.as_f64());
    let bpm_max = bpm_range.and_then(|r| r.get(1)).and_then(|v| v.as_f64());

    match db.search_tracks_filtered(query, artist, genre, bpm_min, bpm_max, key, &tags, limit) {
        Ok(tracks) => {
            let track_data: Vec<Value> = tracks.iter().map(|(t, bpm, musical_key)| {
                json!({
                    "id": t.id,
                    "title": t.title,
                    "artist": t.artist.as_deref().unwrap_or("Unknown"),

                    "rating": t.rating,

                })

fn execute_tag_tracks(db: &Database, input: &Value) -> (String, ActionResult) {
    let track_ids: Vec<i64> = input.get("track_ids")
        .and_then(|v| v.as_array())
        .map(|arr| arr.iter().filter_map(|v| v.as_i64()).collect())
        .unwrap_or_default();
    let add_tags: Vec<String> = input.get("add_tags")
        .and_then(|v| v.as_array())
        .map(|arr| arr.iter().filter_map(|v| v.as_str().map(String::from)).collect())
        .unwrap_or_default();
    let remove_tags: Vec<String> = input.get("remove_tags")
        .and_then(|v| v.as_array())
        .map(|arr| arr.iter().filter_map(|v| v.as_str().map(String::from)).collect())
        .unwrap_or_default();
    let category = input.get("category").and_then(|v| v.as_str()).unwrap_or("general");

    let mut tags_added = 0;
    let mut tags_removed = 0;

    // Add tags
    for tag_name in &add_tags {
        // Ensure tag exists
        let tag_id = match db.get_or_create_tag(tag_name, category) {
            Ok(id) => id,
            Err(_) => continue,
        };
        for track_id in &track_ids {
            if db.add_tag_to_track(*track_id, tag_id).is_ok() {
                tags_added += 1;
            }
        }
    }

    // Remove tags
    for tag_name in &remove_tags {
        if let Ok(Some(tag_id)) = db.get_tag_by_name(tag_name) {
            for track_id in &track_ids {
                if db.remove_tag_from_track(*track_id, tag_id).is_ok() {
                    tags_removed += 1;
                }
            }
        }
    }

    let tool_result = json!({
        "updated_count": track_ids.len(),
        "tags_added": tags_added,
        "tags_removed": tags_removed,
    }).to_string();
    let result = ActionResult {
        tool_name: "tag_tracks".to_string(),
        success: true,
        summary: format!("Updated {} tracks: {} tags added, {} removed", track_ids.len(), tags_added, tags_removed),
        data: Some(json!({ "updated_count": track_ids.len(), "tags_added": tags_added, "tags_removed": tags_removed })),
    };
    (tool_result, result)
}

/// Assemble the full system prompt for ai_chat_v2
pub fn assemble_system_prompt(
    db: &Database,
    taste_profile_json: Option<&str>,
    session_context: Option<&SessionContext>,
) -> String {
    let mut parts: Vec<String> = Vec::new();

    // Layer 1: Identity (always)
    parts.push(SYSTEM_PROMPT.to_string());

    // Layer 2: Taste profile (always, if available)
    if let Some(profile) = taste_profile_json {
        parts.push(format!(
            "\n## User's Music Profile\n\
             The following is a summary of this DJ's music library and listening patterns:\n\
             {}\n\
             Use this to personalize your recommendations and understand their taste.",
            profile
        ));
    }

    // Layer 3: Explicit preferences (always)
    match db.get_all_preferences() {
        Ok(prefs) if !prefs.is_empty() => {
            let mut pref_text = String::from("\n## User's Explicit Preferences\n");
            for (_, category, preference, context) in &prefs {
                pref_text.push_str(&format!("- [{}] {}", category, preference));
                if let Some(ctx) = context {
                    pref_text.push_str(&format!(" (context: {})", ctx));
                }
                pref_text.push('\n');
            }
            pref_text.push_str("\nAlways respect these preferences in your responses and actions.");
            parts.push(pref_text);
        }
        _ => {}
    }

    // Layer 4: Current session (always, if available)
    if let Some(session) = session_context {
        let mut session_text = String::from("\n## Current Session\n");
        if let Some(track) = &session.now_playing {
            session_text.push_str(&format!(
                "Now playing: \"{}\" by {} (BPM: {}, Key: {})\n",
                track.title,
                track.artist.as_deref().unwrap_or("Unknown"),
                track.bpm.map(|b| format!("{:.0}", b)).unwrap_or_else(|| "?".to_string()),
                track.key.as_deref().unwrap_or("?"),
            ));
        }
        if !session.recent_queue.is_empty() {
            session_text.push_str("Recent queue: ");
            let queue_items: Vec<String> = session.recent_queue.iter()
                .map(|t| format!("\"{}\" by {}", t.title, t.artist.as_deref().unwrap_or("Unknown")))
                .collect();
            session_text.push_str(&queue_items.join(", "));
            session_text.push('\n');
        }
        if let Some(playlist_id) = session.active_playlist_id {
            session_text.push_str(&format!("Active playlist ID: {}\n", playlist_id));
        }
        parts.push(session_text);
    }

    // Tool instructions
    parts.push(
        "\n## Tool Usage Guidelines\n\
         - Use search_library to find tracks before creating playlists or queuing\n\
         - When the user asks to play something, search first then queue_tracks\n\
         - When the user expresses a preference, use save_preference to remember it\n\
         - Use recall_conversations when the user references something from a past chat\n\
         - Always explain what you did after taking actions\n\
         - For playlists, aim for smooth BPM transitions and Camelot key compatibility"
        .to_string()
    );

    parts.join("\n")
}

use crate::ai::claude_client::{ClaudeClient, ContentBlockV2, ToolMessage, ToolMessageContent};
use crate::ai::context_assembler::{assemble_system_prompt, SessionContext};
use crate::ai::tool_definitions::get_tool_definitions;
use crate::ai::tool_executor::{execute_tool, ActionResult};
use crate::ai::taste_profile::build_taste_profile;
use crate::db::Database;
use serde::{Deserialize, Serialize};

const MAX_TOOL_ROUNDS: usize = 5;

#[derive(Serialize, Deserialize, Debug, Clone)]
pub struct ChatV2Response {
    pub text: String,
    pub actions: Vec<ActionResult>,
}

/// Run the full tool-use orchestration loop
pub async fn orchestrate_chat(
    db: &Database,
    client: &ClaudeClient,
    message: &str,
    conversation_id: &str,
    session_context: Option<&SessionContext>,
    taste_profile_cache: Option<&str>,
) -> Result<ChatV2Response, String> {
    // 1. Assemble system prompt
    let system_prompt = assemble_system_prompt(db, taste_profile_cache, session_context);

    // 2. Load conversation history from DB
    let history = db.get_conversation_messages(conversation_id)
        .map_err(|e| format!("Failed to load conversation: {}", e))?;

    let mut messages: Vec<ToolMessage> = Vec::new();

    // Add conversation history (last 20 messages to stay within token budget)
    let history_window: Vec<_> = if history.len() > 20 {
        history[history.len() - 20..].to_vec()
    } else {
        history
    };
    for msg in &history_window {
        messages.push(ToolMessage {
            role: msg.role.clone(),
            content: ToolMessageContent::Text(msg.content.clone()),
        });
    }

    // Add current user message
    messages.push(ToolMessage {
        role: "user".to_string(),
        content: ToolMessageContent::Text(message.to_string()),
    });

    // 3. Save user message to DB
    let _ = db.create_message(conversation_id, "user", message);

    // 4. Get tool definitions
    let tools = get_tool_definitions();

    // 5. Tool-use loop
    let mut all_actions: Vec<ActionResult> = Vec::new();
    let mut final_text = String::new();

    for _round in 0..MAX_TOOL_ROUNDS {
        let response = client.chat_with_tools(&system_prompt, messages.clone(), &tools).await?;

        // Process response content blocks
        let mut text_parts: Vec<String> = Vec::new();
        let mut tool_calls: Vec<(String, String, serde_json::Value)> = Vec::new(); // (id, name, input)

        for block in &response.content {
            match block {
                ContentBlockV2::Text { text } => {
                    text_parts.push(text.clone());
                }
                ContentBlockV2::ToolUse { id, name, input } => {
                    tool_calls.push((id.clone(), name.clone(), input.clone()));
                }
                _ => {}
            }
        }

        final_text = text_parts.join("\n");

        // If no tool calls, we're done
        if tool_calls.is_empty() || response.stop_reason == "end_turn" {
            break;
        }

        // Execute tool calls and build tool results
        let mut assistant_content: Vec<ContentBlockV2> = Vec::new();

        // Include text blocks from assistant response
        for block in &response.content {
            match block {
                ContentBlockV2::Text { text } => {
                    assistant_content.push(ContentBlockV2::Text { text: text.clone() });
                }
                ContentBlockV2::ToolUse { id, name, input } => {
                    assistant_content.push(ContentBlockV2::ToolUse {
                        id: id.clone(),
                        name: name.clone(),
                        input: input.clone(),
                    });
                }
                _ => {}
            }
        }

        // Add assistant message with tool_use blocks
        messages.push(ToolMessage {
            role: "assistant".to_string(),
            content: ToolMessageContent::Blocks(assistant_content),
        });

        // Execute tools and add results
        let mut tool_results: Vec<ContentBlockV2> = Vec::new();
        for (tool_id, tool_name, tool_input) in &tool_calls {
            let (tool_result_text, action_result) = execute_tool(db, tool_name, tool_input);
            all_actions.push(action_result);
            tool_results.push(ContentBlockV2::ToolResult {
                tool_use_id: tool_id.clone(),
                content: tool_result_text,
                is_error: None,
            });
        }

        // Add user message with tool results
        messages.push(ToolMessage {
            role: "user".to_string(),
            content: ToolMessageContent::Blocks(tool_results),
        });
    }

    // 6. Save assistant response to DB (with action metadata)
    let metadata = if all_actions.is_empty() {
        None
    } else {
        serde_json::to_string(&all_actions).ok()
    };
    let _ = db.create_message_with_metadata(conversation_id, "assistant", &final_text, metadata.as_deref());

    Ok(ChatV2Response {
        text: final_text,
        actions: all_actions,
    })
}
```

use crate::ai::{orchestrate_chat, ChatV2Response, SessionContext, build_taste_profile};

#[tauri::command]
pub async fn ai_chat_v2(
    state: tauri::State<'_, AppState>,
    message: String,
    conversation_id: String,
    session_context: Option<SessionContext>,
) -> Result<ChatV2Response, AppError> {
    let db_lock = state.db.lock().map_err(|e| AppError::Internal(e.to_string()))?;
    let db = db_lock.as_ref().ok_or(AppError::Internal("Database not initialized".to_string()))?;

    // Get API key
    let api_key = db.get_setting("ai_api_key")
        .map_err(|e| AppError::Database(e.to_string()))?
        .ok_or(AppError::AiNoApiKey)?;

    let client = crate::ai::ClaudeClient::new(api_key);

    // Get or rebuild taste profile
    let taste_profile = {
        let mut cache = state.taste_profile_cache.lock()
            .map_err(|e| AppError::Internal(e.to_string()))?;
        if cache.is_none() {
            if let Ok(profile_json) = build_taste_profile(db) {
                let _ = db.save_taste_profile(&profile_json);
                *cache = Some(profile_json);
            }
        }
        cache.clone()
    };

    // Run orchestrator
    let response = orchestrate_chat(
        db,
        &client,
        &message,
        &conversation_id,
        session_context.as_ref(),
        taste_profile.as_deref(),
    ).await.map_err(|e| AppError::AiNetwork(e))?;

    Ok(response)
}
```

  const state = get();
  const message = state.chatHistory[state.chatHistory.length - 1];

applyQueueAction: (tracks: Track[], mode: string) => {
  const state = get();
  if (mode === 'play_now') {
    set({
      queue: tracks,
      originalQueue: tracks,
      currentTrackIndex: 0,
      currentTrack: tracks[0] || null,
    });
  } else if (mode === 'play_next') {
    const newQueue = [
      ...state.queue.slice(0, state.currentTrackIndex + 1),
      ...tracks,
      ...state.queue.slice(state.currentTrackIndex + 1),
    ];
    set({ queue: newQueue, originalQueue: newQueue });
  } else {
    // append
    const newQueue = [...state.queue, ...tracks];
    set({ queue: newQueue, originalQueue: newQueue });
  }
},
